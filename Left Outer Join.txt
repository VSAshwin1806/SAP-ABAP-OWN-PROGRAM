REPORT zprg_left_outer_join.


" Declared a parameter for user to give input to data
PARAMETERS: p_ebeln TYPE ekko-ebeln.



TYPES: BEGIN OF ty_purch,
ebeln TYPE ekko-ebeln,
bukrs TYPE ekko-bukrs,
bstyp TYPE ekko-bstyp,
bsart TYPE ekko-bsart,
aedat TYPE ekko-aedat,
ernam TYPE ekko-ernam,
lifnr TYPE ekko-lifnr,
ebelp TYPE ekpo-ebelp,
matnr TYPE ekpo-matnr,
menge TYPE ekpo-menge,
meins TYPE ekpo-meins,
END OF ty_purch.


DATA : lt_purch TYPE STANDARD TABLE OF ty_purch.


SELECT
  a~ebeln
  a~bukrs
  a~bstyp
  a~bsart
  a~aedat
  a~ernam
  a~lifnr
  b~ebelp
  b~matnr
  b~menge
  b~meins
  FROM ekko AS a LEFT OUTER JOIN ekpo As b ON " LEFT OUTER JOIN mean it fetch the data from left data of ekko full fields and right table ekpo common field data will come
  a~ebeln = b~ebeln
  INTO TABLE lt_purch
  WHERE a~ebeln = p_ebeln.

      t_table        = lt_purch
