      vbeln

      auart

      waerk

MESSAGE : 'Your are viewing all Sale order Details' TYPE 'I'.

   SELECT
      vbeln

      auart
      netwr
      waerk
    FROM vbak INTO TABLE lt_sales.
     ENDIF.
