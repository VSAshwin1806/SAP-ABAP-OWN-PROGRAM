" Declared a parameter for user to give input and get the data.
PARAMETERS: P_vbeln TYPE vbak-vbeln.



TYPES : BEGIN OF ty_sales,
       vbeln TYPE vbak-vbeln,
       erdat TYPE vbak-erdat,
       ernam TYPE vbak-ernam,
       audat TYPE vbak-audat,
       netwr TYPE vbak-netwr,
       bstnk TYPE vbak-bstnk,
       posnr TYPE vbap-posnr,
       matnr TYPE vbap-matnr,
       matkl TYPE vbap-matkl,
       arktx TYPE vbap-arktx,
       kwmeng TYPE vbap-kwmeng,
  END OF ty_sales.


  DATA : lt_sales TYPE STANDARD TABLE OF ty_sales.


  SELECT
    a~vbeln
    a~erdat
    a~ernam
    a~audat
    a~netwr
    a~bstnk
    b~posnr
    b~matnr
    b~matkl
    b~arktx
    b~kwmeng
    FROM vbak AS a INNER JOIN vbap AS b ON  " INNER JOIN na rendu table la irukkura common field match aana mattum dha data varum.
    a~vbeln = b~vbeln
      INTO TABLE lt_sales
    WHERE a~vbeln = p_vbeln.

        t_table        = lt_sales
    ).
