REPORT z_prg_sales_delivery_invoice.

TABLES: vbak.

SELECTION-SCREEN: BEGIN OF BLOCK b1 WITH FRAME TITLE text-001.
SELECT-OPTIONS: s_vbeln FOR vbak-vbeln.
SELECTION-SCREEN: END OF BLOCK b1.


TYPES: BEGIN OF ty_sales,
    vbeln TYPE vbak-vbeln,
    posnr TYPE vbap-posnr,
    vgbel TYPE lips-vgbel,
    vgpos TYPE lips-vgpos,
    ernam TYPE likp-ernam,
    erdat TYPE likp-erdat,
    vkorg TYPE vbrk-vkorg,
    waerk TYPE vbrk-waerk,
  END OF ty_sales.


DATA: lt_sales TYPE TABLE OF ty_sales,
      lt_sales_final TYPE TABLE OF ty_sales,
      ls_sales TYPE ty_sales.


SELECT
   a~vbeln
   b~posnr
   c~vgbel
   c~vgpos
   c~ernam
   c~erdat
   d~vkorg
   d~waerk
  INTO TABLE lt_sales
  FROM vbak AS a INNER JOIN vbap AS b ON a~vbeln = b~vbeln
  INNER JOIN lips AS c ON c~vgbel = b~vbeln AND c~vgpos = b~posnr
  INNER JOIN likp AS d ON d~vbeln = c~vbeln
  WHERE a~vbeln IN s_vbeln .

LOOP AT lt_sales INTO ls_sales.

  IF ls_sales-waerk IS INITIAL.
    CONTINUE.
 ENDIF.

  APPEND ls_sales TO lt_sales_final.



ENDLOOP.

IF lt_sales_final IS INITIAL.

MESSAGE: 'This Sale order not have the value for the Dcoument Currency' TYPE 'E'.

ENDIF.
