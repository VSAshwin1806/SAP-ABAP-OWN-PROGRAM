TABLES: ekko, ekpo.

TYPES: BEGIN OF ty_po,
         ebeln TYPE ekko-ebeln,
         bukrs TYPE ekko-bukrs,
         lifnr TYPE ekko-lifnr,
         bedat TYPE ekko-bedat,
         ebelp TYPE ekpo-ebelp,
         matnr TYPE ekpo-matnr,
         menge TYPE ekpo-menge,
         netpr TYPE ekpo-netpr,
       END OF ty_po.

DATA: gt_po TYPE TABLE OF ty_po,
      gs_po TYPE ty_po.

PARAMETERS: p_ebeln TYPE ekko-ebeln.

START-OF-SELECTION.

  SELECT a~ebeln
         a~bukrs
         a~lifnr
         a~bedat
         b~ebelp
         b~matnr
         b~menge
         b~netpr
    FROM ekko AS a
    INNER JOIN ekpo AS b
    ON a~ebeln = b~ebeln
    INTO TABLE gt_po.
*    WHERE a~ebeln = p_ebeln.

  IF gt_po IS INITIAL.
    WRITE: / 'No data found for this PO'.
    EXIT.
  ENDIF.

  LOOP AT gt_po INTO gs_po.

    WRITE: / gs_po-ebeln,
             gs_po-bukrs,
             gs_po-lifnr,
             gs_po-bedat,
             gs_po-ebelp,
             gs_po-matnr,
             gs_po-menge,
             gs_po-netpr.

  ENDLOOP.
